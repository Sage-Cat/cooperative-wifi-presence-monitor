cmake_minimum_required(VERSION 3.20)
project(cooperative_wifi_presence_monitor VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
option(PRESENCE_BUILD_SERVICE "Build the Linux service and adapters" ON)
option(PRESENCE_STATIC "Build fully static Linux executables" OFF)
option(PRESENCE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(PRESENCE_STATIC AND PRESENCE_SANITIZERS)
  message(FATAL_ERROR "Static linking and sanitizers are separate build configurations")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow
  "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=."
  "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
if(PRESENCE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(PRESENCE_STATIC)
  set(OPENSSL_USE_STATIC_LIBS TRUE)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()
add_library(presence_core src/domain/policy.cpp src/application/monitor.cpp)
target_include_directories(presence_core PUBLIC include)
if(BUILD_TESTING)
  add_executable(presence_core_tests tests/unit/core.cpp)
  target_link_libraries(presence_core_tests PRIVATE presence_core)
  add_test(NAME presence_core_tests COMMAND presence_core_tests)
  add_executable(presence_core_acceptance tests/integration/core_acceptance.cpp)
  target_link_libraries(presence_core_acceptance PRIVATE presence_core)
  add_test(NAME presence_core_acceptance COMMAND presence_core_acceptance)
endif()
if(NOT PRESENCE_BUILD_SERVICE)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The concrete service supports Linux only; use PRESENCE_BUILD_SERVICE=OFF for the portable core")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
set(json_header "${CMAKE_BINARY_DIR}/_deps/include/nlohmann/json.hpp")
if(NOT EXISTS "${json_header}")
  file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/v3.12.0/single_include/nlohmann/json.hpp"
    "${json_header}"
    EXPECTED_HASH SHA256=aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63
    TLS_VERIFY ON STATUS json_download_status)
  list(GET json_download_status 0 json_download_code)
  if(NOT json_download_code EQUAL 0)
    message(FATAL_ERROR "Pinned JSON dependency download failed: ${json_download_status}")
  endif()
endif()
file(SHA256 "${json_header}" json_hash)
if(NOT json_hash STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  message(FATAL_ERROR "Pinned JSON dependency hash mismatch")
endif()
add_library(presence_json INTERFACE)
target_include_directories(presence_json SYSTEM INTERFACE "${CMAKE_BINARY_DIR}/_deps/include")
add_library(presence_adapters src/adapters/contracts.cpp src/adapters/output.cpp)
target_link_libraries(presence_adapters PUBLIC presence_core presence_json PRIVATE OpenSSL::Crypto)
target_compile_definitions(presence_adapters PUBLIC PRESENCE_BUILD_ID="${PROJECT_VERSION}")
add_executable(wifi-presence-monitor app/main.cpp)
target_link_libraries(wifi-presence-monitor PRIVATE presence_adapters Threads::Threads)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME presence_service COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_SOURCE_DIR}/tests/integration/test_service.py"
    "$<TARGET_FILE:wifi-presence-monitor>")
  add_executable(presence_service_tests tests/integration/service_guest.cpp)
  target_link_libraries(presence_service_tests PRIVATE presence_json OpenSSL::Crypto Threads::Threads)
  add_test(NAME presence_service_guest COMMAND presence_service_tests "$<TARGET_FILE:wifi-presence-monitor>")
  set_tests_properties(presence_service presence_service_guest PROPERTIES TIMEOUT 90)
endif()
configure_file(packaging/wifi-presence-monitor.service.in wifi-presence-monitor.service @ONLY)
install(TARGETS wifi-presence-monitor RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${CMAKE_BINARY_DIR}/wifi-presence-monitor.service"
  DESTINATION lib/systemd/system)
install(FILES packaging/THIRD_PARTY_NOTICES.txt packaging/INSTALL_ALLOWLIST.txt
  packaging/OpenSSL-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/cooperative-wifi-presence-monitor)
